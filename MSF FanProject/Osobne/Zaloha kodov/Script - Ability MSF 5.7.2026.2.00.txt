function onOpen() {
  SpreadsheetApp.getUi().createMenu('🚀 MSF Nástroje')
      .addItem('⚡ Aktualizovať VŠETKO (Playable -> Summon)', 'runOptimizedUpdate')
      .addToUi();
}

/**
 * Hlavná funkcia - stiahne zoznam, zoradí (Playable prvé), stiahne detaily a zapíše.
 */
function runOptimizedUpdate() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("DATA");
  
  if (!sheet) {
    SpreadsheetApp.getUi().alert("Chyba: Hárok s názvom 'DATA' neexistuje!");
    return;
  }
  
  if (sheet.getFilter()) {
    sheet.getFilter().remove(); 
  }
  
  ss.toast('Sťahujem zoznam postáv...', 'Krok 1/4', -1);

  // 1. Získanie zoznamu postáv
  var listResponse = UrlFetchApp.fetch("https://api-prod.marvelstrikeforce.com/services/api/getCharacterList?lang=en");
  var allCharacters = JSON.parse(listResponse.getContentText()).data;
  
  // 2. Triedenie: Najprv Playable, potom Summon
  var playables = [];
  var summons = [];

  allCharacters.forEach(function(c) {
    var status = (c.status || "").toLowerCase();
    if (status === "playable") {
      playables.push(c);
    } else if (status === "summon") {
      summons.push(c);
    }
  });

  // Zoradenie podľa abecedy v rámci skupín (pre prehľadnosť)
  playables.sort(function(a, b) { return a.name.localeCompare(b.name); });
  summons.sort(function(a, b) { return a.name.localeCompare(b.name); });

  // Spojenie: Playables idú prvé, Summons idú na koniec
  var activeChars = playables.concat(summons);

  ss.toast('Sťahujem detaily pre ' + activeChars.length + ' postáv (paralelne)...', 'Krok 2/4', -1);

  // 3. Príprava požiadaviek pre dávkové sťahovanie (Batch Fetching)
  var chunkSize = 40; 
  var allRows = [];
  
  for (var i = 0; i < activeChars.length; i += chunkSize) {
    var chunk = activeChars.slice(i, i + chunkSize);
    var requests = chunk.map(function(char) {
      return {
        url: "https://api-prod.marvelstrikeforce.com/services/api/getCharacterbyId?characterId=" + char.id + "&lang=en",
        muteHttpExceptions: true
      };
    });

    // Sťahujeme 40 postáv NARAZ
    var responses = UrlFetchApp.fetchAll(requests);

    // Spracovanie odpovedí z aktuálnej dávky
    responses.forEach(function(res, index) {
      var charBasicInfo = chunk[index];
      var row = processCharacterData(charBasicInfo, res);
      allRows.push(row);
    });
    
    ss.toast('Spracované ' + Math.min(i + chunkSize, activeChars.length) + ' z ' + activeChars.length, 'Krok 2/4');
  }

  // 4. Výmaz starých dát a zápis nových
  ss.toast('Mažem staré a zapisujem nové dáta...', 'Krok 3/4', -1);
  sheet.clear(); // Toto zabezpečí kompletný výmaz pred zápisom
  
  var headers = [["Meno", "ID (Slug)", "API Link", "Traits", "Portrait", "Basic", "Basic ICO", "Special", "Special ICO", "Ultimate", "Ultimate ICO", "Passive", "Passive ICO"]];
  
  // Formátovanie hlavičky
  sheet.getRange(1, 1, 1, headers[0].length)
       .setValues(headers)
       .setFontWeight("bold")
       .setBackground("#cfe2f3")
       .setFontColor("black");

  if (allRows.length > 0) {
    // Zápis všetkých riadkov naraz
    sheet.getRange(2, 1, allRows.length, allRows[0].length).setValues(allRows);
  }

  // 5. Formátovanie a Čistenie
  ss.toast('Aplikujem štýly...', 'Krok 4/4', -1);
  formatSheet(sheet, allRows.length);
  
  ss.toast('✅ Hotovo! Postavy zoradené (Playable -> Summon).', 'ÚSPECH', 5);
}

/**
 * Pomocná funkcia na spracovanie jednej postavy a vytvorenie riadku
 */
function processCharacterData(charInfo, response) {
  var apiLink = "https://api-prod.marvelstrikeforce.com/services/api/getCharacterbyId?characterId=" + charInfo.id + "&lang=en";
  var traits = (charInfo.traits || []).concat(charInfo.eventTraits || []).map(function(t) { return t.name; }).join(", ");
  
  var row = [
    charInfo.name, 
    charInfo.id, 
    apiLink, 
    traits, 
    charInfo.portrait ? '=IMAGE("' + charInfo.portrait + '")' : "" 
  ];

  var abilitiesData = ["", "", "", "", "", "", "", ""]; 

  if (response.getResponseCode() === 200) {
    try {
      var json = JSON.parse(response.getContentText());
      if (json && json.data && json.data.abilityKit) {
        var kit = json.data.abilityKit;
        var types = [kit.basic, kit.special, kit.ultimate, kit.passive];
        
        var tempAbilities = [];
        types.forEach(function(ability) {
          if (!ability) {
            tempAbilities.push("", ""); 
          } else {
            var lvls = ability.levels || {};
            var maxLvl = lvls["8"] || lvls["7"] || lvls["6"] || lvls["5"] || lvls["1"] || ability;
            
            var desc = (maxLvl.description || "").replace(/<[^>]*>/g, ""); 
            var finalDesc = (ability.name || "Unknown").toUpperCase() + "\n\n" + desc;
            var iconFormula = ability.icon ? '=IMAGE("' + ability.icon + '")' : "";
            
            tempAbilities.push(finalDesc, iconFormula);
          }
        });
        abilitiesData = tempAbilities;
      }
    } catch (e) {
      console.error("Chyba JSON: " + charInfo.name, e);
    }
  }
  
  return row.concat(abilitiesData);
}

/**
 * Funkcia na grafickú úpravu tabuľky
 */
function formatSheet(sheet, numRows) {
  if (numRows === 0) return;
  var lastCol = 13; 

  sheet.setColumnWidth(1, 150); 
  sheet.setColumnWidth(2, 120); 
  sheet.setColumnWidth(3, 100); 
  sheet.setColumnWidth(4, 200); 
  
  [6, 8, 10, 12].forEach(function(col) { sheet.setColumnWidth(col, 350); });
  [5, 7, 9, 11, 13].forEach(function(col) { sheet.setColumnWidth(col, 60); });

  var dataRange = sheet.getRange(2, 1, numRows, lastCol);
  dataRange.setWrap(true)
           .setVerticalAlignment("middle")
           .setHorizontalAlignment("left");
           
  [5, 7, 9, 11, 13].forEach(function(col) {
     sheet.getRange(2, col, numRows, 1).setHorizontalAlignment("center");
  });

  sheet.setFrozenRows(1);
  sheet.getRange(1, 1, 1, lastCol).setBackground("#4a86e8").setFontColor("white");
}
